// OPT-1 (Constant Folding):
//   t1 = 2 + 3  ->  t1 = 5
//   t2 = 4 * 5  ->  t2 = 20
//
// OPT-2 (Propagacion de constantes):
//   a = t1  ->  a = 5       (t1 es constante)
//   b = t2  ->  b = 20      (t2 es constante)
//   t3 = b + 1  ->  t3 = 20 + 1   (b es constante)
//   cout << a   ->  cout << 5
//   cout << b   ->  cout << 20
//
// OPT-3 (Eliminacion de codigo muerto):
//   t1 = 5   eliminada  (t1 ya no es usada despues de OPT-2)
//   t2 = 20  eliminada  (t2 ya no es usada despues de OPT-2)

int a = 2 + 3;
int b = 4 * 5;
int c = b;
cout << a;
cout << b;
cout << c;
